def _arrays(seed=0, n=10):

def test_add_noise_scaling_sharing_and_reproducibility():
    X = np.zeros((200, 361))
    assert add_noise(X, 0.0, 3, 0) is X                                              # no noise at multiplier 0
    n1, n4 = add_noise(X, 1.0, 3, 1), add_noise(X, 4.0, 3, 1)
    assert n1.std() == pytest.approx(BASE_NOISE, rel=0.05) and n4.std() / n1.std() == pytest.approx(4.0, rel=1e-9)
    np.testing.assert_array_equal(n1, add_noise(X, 1.0, 3, 1))                       # reproducible
    other = np.random.default_rng(0).normal(size=X.shape)
    np.testing.assert_allclose(add_noise(other, 1.0, 3, 1) - other, n1)               # SAME realisation for every state
    assert not np.allclose(n1, add_noise(X, 1.0, 3, 2)) and not np.allclose(n1, add_noise(X, 1.0, 4, 1))

def _cell(r2, lo=0.05):
    return {"r2": r2, "r2_lo": r2 - 0.05, "r2_hi": r2 + 0.05, "diff_vs_null": {"diff": r2, "lo": lo, "hi": r2 + 0.1}}


def _entry(pool, sub, lo=0.05):
    return {"pool4": {r: _cell(pool, lo) for r in ("linear", "rbf")}, "sub24": {r: _cell(sub, lo) for r in ("linear", "rbf")}}


def _sres(m=0.6, m_sub=0.5, p=0.22, s=0.5, h=0.1, u=0.3, null=-0.01, pc=0.999, mp=0.4, lo=0.05, clip=0.0,
          probe="given", drop=()):
    mults = ("0.0", "1.0", "4.0")
    st = {"mod": {k: _entry(m, m_sub, lo) for k in mults}, "mod_s": {k: _entry(s, s, lo) for k in mults},
          "mod_H": {k: _entry(h, h, lo) for k in mults}, "mod_ring": {k: _entry(s, s, lo) for k in mults},
          "u_full": {k: _entry(u, u, lo) for k in mults}, "u_ring": {k: _entry(u, u, lo) for k in mults},
          "NULL": {k: _entry(null, null) for k in mults}}
    for d in drop:
        st.pop(d)
    return {"states": st, "clip_events": clip, "probe_r2": p if probe == "given" else None,
            "baselines": {"fft_concat": {"shape": {"linear": {"r2": pc}}}, "map": {"position": {"linear": {"r2": mp}}}}}


def test_verdict_invalid_inconclusive_and_the_three_controls():
    assert verdict(_sres(clip=3.0))["outcome"] == "INVALID_CLIPPED"
    assert verdict(_sres(drop=("NULL",)))["outcome"] == "INCONCLUSIVE"
    assert verdict(_sres(pc=0.5))["outcome"] == "POSITIVE_CONTROL_FAILS"
    leaky = _sres(); leaky["states"]["NULL"]["16.0"] = _entry(0.3, -0.01) if "16.0" in leaky["states"]["NULL"] else None
    leaky["states"]["NULL"]["4.0"] = _entry(0.2, -0.01)                                # a leak at ANY noise level counts
    assert verdict(leaky)["outcome"] == "CONTROL_FAILS"
    assert verdict(_sres(null=0.04))["outcome"] != "CONTROL_FAILS"
    assert verdict(_sres(mp=0.1))["outcome"] == "STATE_CONTROL_FAILS"
    assert verdict(_sres(mp=0.26))["outcome"] != "STATE_CONTROL_FAILS"
